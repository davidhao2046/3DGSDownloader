cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(GsdlGraphics LANGUAGES CXX)
  include(CTest)
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Graphics requires Windows x64/MSVC")
endif()
get_filename_component(GSDL_DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party" ABSOLUTE)
foreach(_vendor skia vulkan)
  set(_root "${GSDL_DEPENDENCY_ROOT}/${_vendor}")
  file(READ "${_root}/source-lock.json" _lock)
  string(JSON _count LENGTH "${_lock}" files)
  math(EXPR _last "${_count}-1")
  foreach(_i RANGE 0 ${_last})
    string(JSON _path GET "${_lock}" files ${_i} path)
    string(JSON _expected GET "${_lock}" files ${_i} sha256)
    file(SHA256 "${_root}/${_path}" _actual)
    if(NOT _actual STREQUAL _expected)
      message(FATAL_ERROR "Changed graphics dependency: ${_vendor}/${_path}")
    endif()
  endforeach()
endforeach()
file(READ "${GSDL_DEPENDENCY_ROOT}/dependencies.json" _runtime_lock)
string(JSON _expected_dll_hash GET "${_runtime_lock}" skia sha256)
string(JSON _notice_count LENGTH "${_runtime_lock}" skia notices)
math(EXPR _notice_last "${_notice_count}-1")
foreach(_i RANGE 0 ${_notice_last})
  string(JSON _notice_path GET "${_runtime_lock}" skia notices ${_i} path)
  string(JSON _notice_hash GET "${_runtime_lock}" skia notices ${_i} sha256)
  file(SHA256 "${GSDL_DEPENDENCY_ROOT}/../${_notice_path}" _actual_notice_hash)
  if(NOT _actual_notice_hash STREQUAL _notice_hash)
    message(FATAL_ERROR "Changed SkiaSharp runtime notice: ${_notice_path}")
  endif()
endforeach()
set(GSDL_SKIA_DLL "${GSDL_DEPENDENCY_ROOT}/installed/libSkiaSharp.dll" CACHE FILEPATH "Locked official Skia native DLL")
if(NOT EXISTS "${GSDL_SKIA_DLL}")
  message(FATAL_ERROR "Missing Skia runtime. Run third_party/prepare.ps1 first.")
endif()
file(SHA256 "${GSDL_SKIA_DLL}" _dll_hash)
if(NOT _dll_hash STREQUAL _expected_dll_hash)
  message(FATAL_ERROR "Skia runtime DLL hash differs from dependencies.json")
endif()
add_library(gsdl_graphics STATIC src/skia_canvas.cpp src/win32_presenter.cpp)
target_compile_features(gsdl_graphics PUBLIC cxx_std_20)
target_include_directories(gsdl_graphics PUBLIC include PRIVATE "${GSDL_DEPENDENCY_ROOT}/skia")
target_include_directories(gsdl_graphics SYSTEM PRIVATE "${GSDL_DEPENDENCY_ROOT}/vulkan/include")
target_compile_definitions(gsdl_graphics PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(gsdl_graphics PRIVATE GSDL_SKIA_SHA256="${_expected_dll_hash}")
target_compile_options(gsdl_graphics PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(gsdl_graphics PRIVATE bcrypt user32)

function(gsdl_stage_graphics target)
  get_filename_component(_stage_dependencies "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/../third_party" ABSOLUTE)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${GSDL_SKIA_DLL}" "$<TARGET_FILE_DIR:${target}>/libSkiaSharp.dll"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${_stage_dependencies}/skia/LICENSE" "$<TARGET_FILE_DIR:${target}>/Skia-LICENSE.txt"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${_stage_dependencies}/skia/LICENSE-SkiaSharp.txt" "$<TARGET_FILE_DIR:${target}>/SkiaSharp-LICENSE.txt"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${_stage_dependencies}/skia/THIRD-PARTY-NOTICES.txt" "$<TARGET_FILE_DIR:${target}>/SkiaSharp-THIRD-PARTY-NOTICES.txt")
endfunction()

if(BUILD_TESTING)
  add_executable(gsdl_graphics_tests tests/graphics_tests.cpp)
  target_link_libraries(gsdl_graphics_tests PRIVATE gsdl_graphics)
  target_compile_options(gsdl_graphics_tests PRIVATE /W4 /EHsc /utf-8 /permissive-)
  gsdl_stage_graphics(gsdl_graphics_tests)
  add_test(NAME graphics.skia_vulkan_worker COMMAND gsdl_graphics_tests "${GSDL_SKIA_DLL}")
  set_tests_properties(graphics.skia_vulkan_worker PROPERTIES TIMEOUT 45)
endif()

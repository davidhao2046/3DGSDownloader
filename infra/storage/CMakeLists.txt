cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(gsdl_storage_checks LANGUAGES CXX)
  include(../../cmake/MsvcUtf8Includes.cmake)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
  add_compile_options(/W4 /EHsc /utf-8 /permissive-)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_library(gsdl_domain STATIC ../../core/domain/plan.cpp)
  target_include_directories(gsdl_domain PUBLIC ../..)
endif()
add_library(gsdl_storage STATIC plan_codec.cpp plan_storage.cpp)
target_include_directories(gsdl_storage PUBLIC ../..)
target_compile_features(gsdl_storage PUBLIC cxx_std_20)
target_link_libraries(gsdl_storage PUBLIC gsdl_domain PRIVATE bcrypt)
if(BUILD_TESTING)
  add_executable(storage_tests ../../tests/storage_tests.cpp)
  target_link_libraries(storage_tests PRIVATE gsdl_storage)
  add_test(NAME storage.strict_atomic_async COMMAND storage_tests)
endif()

cmake_minimum_required(VERSION 3.25)
project(3DGSDownloader VERSION 0.1.0 LANGUAGES CXX)
include(cmake/MsvcUtf8Includes.cmake)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_options(/W4 /EHsc /utf-8 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
include(CTest)
add_library(gsdl_domain STATIC core/domain/plan.cpp)
target_include_directories(gsdl_domain PUBLIC .)
add_library(gsdl_application STATIC core/application/plan_service.cpp)
target_include_directories(gsdl_application PUBLIC .)
target_link_libraries(gsdl_application PUBLIC gsdl_domain)
add_library(gsdl_transfer STATIC infra/net/http.cpp adapters/sources/insta360.cpp core/application/download.cpp core/application/transfer_service.cpp)
target_include_directories(gsdl_transfer PUBLIC .)
target_link_libraries(gsdl_transfer PUBLIC gsdl_application winhttp bcrypt)
add_subdirectory(infra/storage)
add_library(gsdl_task_list STATIC core/application/task_list.cpp)
target_link_libraries(gsdl_task_list PUBLIC gsdl_transfer)
add_library(gsdl_task_workspace STATIC infra/storage/task_workspace.cpp)
target_link_libraries(gsdl_task_workspace PUBLIC gsdl_task_list gsdl_storage)
if(BUILD_TESTING)
 add_executable(task_list_tests tests/task_list_tests.cpp)
 target_link_libraries(task_list_tests PRIVATE gsdl_task_list)
 add_test(NAME tasks.queue_ownership_restore COMMAND task_list_tests)
 add_executable(task_workspace_tests tests/task_workspace_tests.cpp)
 target_link_libraries(task_workspace_tests PRIVATE gsdl_task_workspace)
 add_test(NAME workspace.strict_atomic_async COMMAND task_workspace_tests "${CMAKE_BINARY_DIR}/task-workspace-test-data")
endif()
add_subdirectory(graphics)
add_subdirectory(ui/text)
if(BUILD_TESTING)
 add_executable(plan_tests tests/plan_tests.cpp)
 target_link_libraries(plan_tests PRIVATE gsdl_application)
 add_test(NAME plan.model_history_save_contracts COMMAND plan_tests)
 add_executable(text_tests tests/text_tests.cpp)
 target_link_libraries(text_tests PRIVATE gsdl_text)
 add_test(NAME text.editor_and_cjk COMMAND text_tests)
 add_executable(transfer_tests tests/transfer_tests.cpp)
 target_link_libraries(transfer_tests PRIVATE gsdl_transfer gsdl_storage ws2_32)
 add_test(NAME transfer.source_http_files_cancel COMMAND transfer_tests "${CMAKE_BINARY_DIR}/transfer-test-data")
 set_tests_properties(transfer.source_http_files_cancel PROPERTIES TIMEOUT 60)
endif()
add_executable(3DGSDownloader WIN32 apps/desktop/main.cpp)
target_include_directories(3DGSDownloader PRIVATE .)
target_link_libraries(3DGSDownloader PRIVATE gsdl_graphics gsdl_text gsdl_application gsdl_storage gsdl_transfer gsdl_task_list gsdl_task_workspace user32 shell32 imm32 dwmapi ole32 comctl32)
if(EXISTS "${CMAKE_SOURCE_DIR}/ui/accessibility.cpp")
 target_sources(3DGSDownloader PRIVATE ui/accessibility.cpp)
 target_link_libraries(3DGSDownloader PRIVATE uiautomationcore oleaut32)
endif()
gsdl_stage_graphics(3DGSDownloader)
add_executable(3DGSDownloaderCli apps/cli/main.cpp apps/cli/transfer_cli.cpp)
target_link_libraries(3DGSDownloaderCli PRIVATE gsdl_application gsdl_storage gsdl_transfer)
if(BUILD_TESTING)
 find_program(GSDL_POWERSHELL NAMES pwsh pwsh.exe REQUIRED)
 add_test(NAME cli.plan_roundtrip COMMAND "${GSDL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/tests/cli_integration.ps1" -Executable "$<TARGET_FILE:3DGSDownloaderCli>")
 set_tests_properties(cli.plan_roundtrip PROPERTIES TIMEOUT 60)
endif()

if(BUILD_TESTING)
 add_executable(task_visual_tests tests/task_visual_tests.cpp)
 target_link_libraries(task_visual_tests PRIVATE gsdl_graphics gsdl_text gsdl_storage ole32)
 add_test(NAME ui.task_visual_matrix COMMAND task_visual_tests "${GSDL_SKIA_DLL}" "${CMAKE_BINARY_DIR}/visual-matrix")
 add_executable(component_visual_tests tests/component_visual_tests.cpp)
 target_link_libraries(component_visual_tests PRIVATE gsdl_graphics gsdl_text gsdl_storage ole32)
 add_test(NAME ui.component_state_matrix COMMAND component_visual_tests "${GSDL_SKIA_DLL}" "${CMAKE_BINARY_DIR}/component-matrix")
 add_executable(page_visual_tests tests/page_visual_tests.cpp)
 target_link_libraries(page_visual_tests PRIVATE gsdl_graphics gsdl_text gsdl_storage ole32)
 add_test(NAME ui.page_visual_matrix COMMAND page_visual_tests "${GSDL_SKIA_DLL}" "${CMAKE_BINARY_DIR}/page-matrix")
endif()
